private const string CouponCodeIndexName = "ux_couponCode"; private const int DuplicateKeyErrorCode = 11000; private readonly IMongoCollection _collection; public MongoCouponRepository(string connectionString, string databaseName, string collectionName) { var client = new MongoClient(connectionString); var database = client.GetDatabase(databaseName); _collection = database.GetCollection(collectionName); } /// /// Point 14: use the collection if it already exists, otherwise create it - then make sure /// the unique index on couponCode exists either way. Safe to call on every startup /// (CreateOneAsync/CreateCollection are no-ops when the target already exists in the way we need). /// public async Task EnsureCollectionAndIndexesAsync() { var database = _collection.Database; var existingNames = await (await database.ListCollectionNamesAsync( new ListCollectionNamesOptions { Filter = new MongoDB.Bson.BsonDocument("name", _collection.CollectionNamespace.CollectionName) })).ToListAsync(); if (existingNames.Count == 0) { await database.CreateCollectionAsync(_collection.CollectionNamespace.CollectionName); Log.Information("Created MongoDB collection {Collection}", _collection.CollectionNamespace.CollectionName); } var indexKeys = Builders.IndexKeys.Ascending(c => c.CouponCode); var indexModel = new CreateIndexModel( indexKeys, new CreateIndexOptions { Unique = true, Name = CouponCodeIndexName }); await _collection.Indexes.CreateOneAsync(indexModel); } /// /// Fast existence check for a coupon code, backed by the unique index - stays cheap even /// with 10 lakh+ documents since it's an indexed equality lookup, not a collection scan. /// public async Task CouponExistsAsync(string couponCode) { var filter = Builders.Filter.Eq(c => c.CouponCode, couponCode); var found = await _collection.Find(filter).Limit(1).AnyAsync(); return found; } /// /// Inserts the document. Returns false (instead of throwing) on a duplicate-key race, so the /// caller can just regenerate and retry rather than special-case a Mongo exception type. /// public async Task TryInsertAsync(CouponDocument document) { try { await _collection.InsertOneAsync(document); return true; } catch (MongoWriteException ex) when (ex.WriteError?.Category == ServerErrorCategory.DuplicateKey) { Log.Warning("Duplicate coupon code on insert, will retry: {CouponCode}", document.CouponCode); return false; } }
