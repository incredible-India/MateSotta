public const int CodeLength = 16; private const string DigitChars = "0123456789"; private const string AlphaNumericChars = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"; public static string Generate() { var sb = new StringBuilder(CodeLength); using (var rng = RandomNumberGenerator.Create()) { sb.Append(DigitChars[GetRandomIndex(rng, DigitChars.Length)]); for (var i = 1; i < CodeLength; i++) { sb.Append(AlphaNumericChars[GetRandomIndex(rng, AlphaNumericChars.Length)]); } } return sb.ToString(); } /// /// Unbiased random index in [0, exclusiveUpperBound) via rejection sampling on a single byte. /// exclusiveUpperBound must be <= 256 (true for both our alphabets). /// private static int GetRandomIndex(RandomNumberGenerator rng, int exclusiveUpperBound) { if (exclusiveUpperBound <= 0 || exclusiveUpperBound > 256) { throw new ArgumentOutOfRangeException(nameof(exclusiveUpperBound)); } // Largest multiple of exclusiveUpperBound that fits in a byte (0-255); values at or // above this are discarded and re-rolled so every valid index has exactly equal odds. var limit = (256 / exclusiveUpperBound) * exclusiveUpperBound; var buffer = new byte[1]; int sample; do { rng.GetBytes(buffer); sample = buffer[0]; } while (sample >= limit); return sample % exclusiveUpperBound; } }
