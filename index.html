public static void Configure(AppSettings settings) { Log.Logger = new LoggerConfiguration() .MinimumLevel.Information() .WriteTo.File( path: settings.LogFilePath, rollingInterval: RollingInterval.Day, rollOnFileSizeLimit: true, fileSizeLimitBytes: settings.LogFileSizeLimitMB * 1024L * 1024L, retainedFileCountLimit: settings.LogRetainedFileCount, shared: true, outputTemplate: "{Timestamp:yyyy-MM-dd HH:mm:ss.fff zzz} [{Level:u3}] {Message:lj}{NewLine}{Exception}") .Enrich.FromLogContext() .CreateLogger(); }
